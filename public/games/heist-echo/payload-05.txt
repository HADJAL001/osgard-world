      kiteToExit:kite.has(exitKey)?kite.get(exitKey):null,
      reachableCaseApproaches:reachableCase.length,
      patrolBlocked:(mission.guardRoute||[]).length-patrolOpen,
      patrolOpen,
      blockedCells:walls.size
    };
  }

  function forgeHash(value) {
    let hash = 0x811c9dc5;
    for (let i = 0; i < value.length; i += 1) { hash ^= value.charCodeAt(i); hash = Math.imul(hash, 0x01000193); }
    return ('00000000' + (hash >>> 0).toString(16)).slice(-8);
  }
  function forgePayload(input) {
    if (!input || typeof input !== 'object' || Array.isArray(input)) throw new Error('Нужен объект испытания.');
    const cleanName = String(input.name || '').replace(/[\u0000-\u001f\u007f]/g, '').trim().slice(0, 30);
    if (cleanName.length < 3) throw new Error('Название должно содержать от 3 до 30 символов.');
    const allowedTicks = [18,20,22,24,26,28,30,32,34,36,38,40];
    const maxTicks = Number(input.maxTicks);
    if (!allowedTicks.includes(maxTicks)) throw new Error('Лимит тактов должен быть от 18 до 40.');
    let noDecoy = input.noDecoy === true; const noHack = input.noHack === true, noWarnings = input.noWarnings === true;
    const guardSpeed = Number(input.guardSpeed) === 2 ? 2 : Number(input.guardSpeed) === 1 ? 1 : NaN;
    if (!Number.isFinite(guardSpeed)) throw new Error('Некорректная скорость патруля.');
    const capRaw = input.maxCommands === null || input.maxCommands === undefined || input.maxCommands === '' ? null : Number(input.maxCommands);
    const maxCommands = capRaw === null ? null : capRaw;
    if (maxCommands !== null && ![18,22,26,30].includes(maxCommands)) throw new Error('Лимит команд должен быть 18, 22, 26 или 30.');
    const terrainInput = Array.isArray(input.terrainWalls) ? input.terrainWalls : [];
    if (terrainInput.length > 10) throw new Error('На карте можно разместить не более 10 новых укреплений.');
    const terrainWalls = [];
    const terrainKeys = new Set();
    for (const raw of terrainInput) {
      const x = Array.isArray(raw) ? Number(raw[0]) : Number(raw && raw.x);
      const y = Array.isArray(raw) ? Number(raw[1]) : Number(raw && raw.y);
      if (!Number.isInteger(x) || !Number.isInteger(y) || x < 1 || x > 10 || y < 1 || y > 6) throw new Error('Укрепление должно находиться внутри карты, не на внешней границе.');
      const key = `${x},${y}`;
      if (terrainKeys.has(key)) throw new Error('В списке укреплений есть повторяющаяся клетка.');
      if (typeof terrainCellReserved === 'function' && terrainCellReserved(x, y)) throw new Error(`Клетка ${x + 1}:${y + 1} занята обязательным объектом или маршрутом патруля.`);
      terrainKeys.add(key); terrainWalls.push([x,y]);
    }
    terrainWalls.sort((a,b) => a[1]-b[1] || a[0]-b[0]);
    const topology = analyzeTerrain(terrainWalls);
    if (!topology.valid) throw new Error('Карта не проходит анализ маршрутов: ' + topology.errors.join(' '));
    const hasModifier = maxTicks < 40 || noDecoy || noHack || noWarnings || guardSpeed > 1 || maxCommands !== null || terrainWalls.length > 0;
    if (!hasModifier) throw new Error('Добавь хотя бы одно ограничение или измени геометрию карты.');
    return { schema:1, game:'HEIST//ECHO', name:cleanName, maxTicks, noDecoy, noHack, noWarnings, guardSpeed, maxCommands, terrainWalls };
  }
  function forgeCanonical(payload) {
    return JSON.stringify({ schema:1, game:'HEIST//ECHO', name:payload.name, maxTicks:payload.maxTicks, noDecoy:payload.noDecoy, noHack:payload.noHack, noWarnings:payload.noWarnings, guardSpeed:payload.guardSpeed, maxCommands:payload.maxCommands, terrainWalls:payload.terrainWalls || [] });
  }
  function makeForgeContract(input) {
    const payload = forgePayload(input), canonical = forgeCanonical(payload), id = 'forge-' + forgeHash(canonical);
    const rules = [`Окно извлечения: не более ${payload.maxTicks} тактов.`];
    if (payload.noDecoy) rules.push('Шумовая приманка полностью отключена.');
    if (payload.noHack) rules.push('Команда HACK запрещена; камеру придётся обходить.');
    if (payload.noWarnings) rules.push('Ноль предупреждений за всю операцию.');
    if (payload.guardSpeed === 2) rules.push('Патруль перемещается на две позиции за такт.');
    if (payload.maxCommands !== null) rules.push(`Лимит общего плана: ${payload.maxCommands} команд.`);
    const modifiers = [];
    if (payload.noDecoy) modifiers.push('без приманки');
    if (payload.noHack) modifiers.push('без взлома');
    if (payload.noWarnings) modifiers.push('чистый журнал');
    if (payload.guardSpeed === 2) modifiers.push('ускоренный патруль');
    if (payload.maxCommands !== null) modifiers.push(`${payload.maxCommands} команд`);
    if (payload.terrainWalls.length) modifiers.push(`${payload.terrainWalls.length} укрепл. на карте`);
    const contract = { ...payload, terrainWalls:payload.terrainWalls || [], id, number:'LAB', short:'Авторское испытание', brief:`Пользовательский контракт: ${modifiers.concat([`дедлайн T+${payload.maxTicks}`]).join(' · ')}. Все правила проверяются после реального прогона симуляции.`, rules, mode:'forge', custom:true, createdAt:typeof input.createdAt === 'string' ? input.createdAt : new Date().toISOString() };
    if (payload.terrainWalls.length) rules.push(`Архитектура: ${payload.terrainWalls.length} укреплений добавлены на карту; маршрут должен обходить их.`);
    return contract;
  }
  function encodeForgeCode(contract) {
    const payload = forgePayload(contract.payload || contract), raw = forgeCanonical(payload);
    const bytes = new TextEncoder().encode(raw); let binary = '';
    for (let i = 0; i < bytes.length; i += 0x8000) binary += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
    const encoded = btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/g, '');
    return `HE2F-${encoded}-${forgeHash(raw)}`;
  }
  function decodeForgeCode(value) {
    const code = String(value || '').trim().replace(/\s+/g, '');
    if (code.length > 4096) throw new Error('Код слишком длинный.');
    const match = /^HE2F-([A-Za-z0-9_-]+)-([0-9a-fA-F]{8})$/.exec(code);
    if (!match) throw new Error('Формат кода неверен. Нужен полный код HE2F-…');
    const base64 = match[1].replace(/-/g, '+').replace(/_/g, '/') + '='.repeat((4 - match[1].length % 4) % 4);
    let binary; try { binary = atob(base64); } catch { throw new Error('Код не декодируется.'); }
    const bytes = Uint8Array.from(binary, (ch) => ch.charCodeAt(0));
    let raw; try { raw = new TextDecoder('utf-8', { fatal:true }).decode(bytes); } catch { throw new Error('Код содержит повреждённый текст.'); }
    if (forgeHash(raw).toLowerCase() !== match[2].toLowerCase()) throw new Error('Контрольная сумма не совпадает. Код повреждён.');
    let parsed; try { parsed = JSON.parse(raw); } catch { throw new Error('В коде нет корректного JSON.'); }
    if (!parsed || parsed.schema !== 1 || parsed.game !== 'HEIST//ECHO') throw new Error('Эта версия кода испытания не поддерживается.');
    return makeForgeContract(parsed);
  }

  function scoreOperation(state) {
    const commands = plannedCommands(state), total = commands.length;
    const moveCount = commands.filter((c) => c.type === Game.ACTION.MOVE).length;
    const decoyCommands = commands.filter((c) => c.type === Game.ACTION.DECOY).length;
    const decoyUses = (state.events || []).filter((e) => e.kind === 'decoy').length;
    const warningCount = (state.events || []).filter((e) => e.kind === 'warning').length;
    const dangerCount = (state.events || []).filter((e) => e.kind === 'danger').length;
    const mira = (state.agents || []).find((a) => a.id === 'mira');
    const kite = (state.agents || []).find((a) => a.id === 'kite');
    const coordinated = !!(mira && mira.plan.length && kite && kite.plan.length);
    const hasHack = (state.events || []).some((e) => /MIRA disabled the camera/i.test(String(e.message)));
    const won = state.status === 'won';
    const ticks = Math.max(0, Number(state.tick) || 0);
    const pace = Math.round(250 * Math.max(0, 1 - ticks / Game.MAX_TICKS));
    const stealth = Math.max(0, 180 - warningCount * 55 - dangerCount * 75 - decoyUses * 35);
    const economy = Math.max(0, 120 - Math.max(0, total - 18) * 5 - Math.max(0, moveCount - 17) * 2);
    const coordination = (coordinated ? 50 : 0) + (hasHack ? 30 : 0);
    const cleanBonus = decoyCommands === 0 && warningCount === 0 ? 40 : 0;
    const score = won ? Math.max(0, Math.min(1000, 450 + pace + stealth + economy + coordination + cleanBonus)) : Math.max(0, Math.min(449, 100 + Math.round(pace*.25) - warningCount*20));
    return { score, ticks, totalCommands:total, moveCount, warningCount, dangerCount, decoyCommands, decoyUses, coordinated, hasHack, won };
  }
  function contractChecks(contract, state, metrics) {
    const checks = [];
    if (!metrics.won) {
      checks.push({ pass:false, title:'Извлечение', detail:state.result && state.result.reason ? String(state.result.reason) : 'Операция не завершилась успешно.' });
      return checks;
    }
    checks.push({ pass:true, title:'Носитель доставлен', detail:`Успешное извлечение подтверждено на T+${String(metrics.ticks).padStart(2,'0')}.` });
    if (contract.custom) {
      const commands = plannedCommands(state);
      checks.push({ pass:metrics.ticks <= contract.maxTicks, title:'Окно извлечения', detail:`T+${metrics.ticks} / лимит ${contract.maxTicks}.` });
      if (contract.noDecoy) checks.push({ pass:metrics.decoyCommands === 0, title:'Без приманки', detail:metrics.decoyCommands === 0 ? 'В плане нет команды приманки.' : `Команд приманки: ${metrics.decoyCommands}.` });
      if (contract.noHack) { const count = commands.filter((c) => c.type === Game.ACTION.HACK).length; checks.push({ pass:count === 0, title:'Протокол без взлома', detail:count === 0 ? 'Команда HACK не использовалась.' : `Найдено команд HACK: ${count}.` }); }
      if (contract.noWarnings) checks.push({ pass:metrics.warningCount === 0, title:'Чистый журнал', detail:metrics.warningCount === 0 ? 'Предупреждений нет.' : `Предупреждений: ${metrics.warningCount}.` });
      if (contract.guardSpeed > 1) checks.push({ pass:true, title:'Ускоренный патруль', detail:'Извлечение подтверждено при ускоренном патруле.' });
      if (contract.maxCommands !== null) checks.push({ pass:metrics.totalCommands <= contract.maxCommands, title:'Бюджет команд', detail:`${metrics.totalCommands} / ${contract.maxCommands} команд.` });
      if (contract.terrainWalls && contract.terrainWalls.length) { const pathCommands=plannedCommands(state).filter(c=>c.type===Game.ACTION.MOVE); const intersects=pathCommands.some(c=>contract.terrainWalls.some(p=>p[0]===c.x&&p[1]===c.y)); const blocked=Number(state.__architect&&state.__architect.blockedCommands)||0; checks.push({pass:!intersects&&blocked===0,title:'Геометрия BLACKSITE',detail:blocked?`Движения остановлены геометрией: ${blocked}.`:(intersects?'Маршрут пересёк новую геометрию.':'Ни одна команда движения не пересекает пользовательские укрепления.')}); }
      return checks;
    }
    if (contract.id === 'ghost') {
      checks.push({ pass:metrics.decoyCommands === 0, title:'Без приманки', detail:metrics.decoyCommands === 0 ? 'В план не добавлено ни одной приманки.' : `В плане осталось команд приманки: ${metrics.decoyCommands}.` });
      checks.push({ pass:metrics.warningCount === 0, title:'Чистый журнал', detail:metrics.warningCount === 0 ? 'Предупреждений нет.' : `Предупреждений: ${metrics.warningCount}.` });
      checks.push({ pass:metrics.ticks <= 28, title:'Окно извлечения', detail:`T+${metrics.ticks} / лимит 28.` });
    } else if (contract.id === 'signal-lock') {
      const earlyHack = hasCameraHackByMira(state);
      checks.push({ pass:earlyHack, title:'Протокол MIRA', detail:earlyHack ? 'Камера отключена MIRA не позднее T+05.' : 'В журнале нет отключения камеры MIRA к T+05.' });
    } else if (contract.id === 'redline') {
      checks.push({ pass:true, title:'Ускоренный патруль', detail:'Извлечение подтверждено при ускоренном продвижении охраны.' });
    } else if (contract.id === 'zero-hour') {
      checks.push({ pass:metrics.ticks <= 22, title:'Жёсткий дедлайн', detail:`T+${metrics.ticks} / лимит 22.` });
    } else if (contract.id === 'clean-room') {
      checks.push({ pass:metrics.decoyCommands === 0, title:'Ноль приманок', detail:metrics.decoyCommands === 0 ? 'Приманка не включена в план.' : `В очередях команд приманки: ${metrics.decoyCommands}.` });
      checks.push({ pass:metrics.warningCount === 0, title:'Ноль предупреждений', detail:metrics.warningCount === 0 ? 'Журнал чист.' : `Предупреждений: ${metrics.warningCount}.` });
      checks.push({ pass:metrics.ticks <= 26, title:'Окно чистого выхода', detail:`T+${metrics.ticks} / лимит 26.` });
    }
    return checks;
  }

  function evaluateAchievements(contract, metrics, checks, dailyId, contractCleared) {
    const unlocked = [];
    const tryUnlock = (id, condition) => { if (condition && unlockAchievement(id)) unlocked.push(ACHIEVEMENTS.find((a) => a.id === id)); };
    tryUnlock('first-breach', metrics.won);
    tryUnlock('ghost-in', metrics.won && metrics.warningCount === 0 && metrics.decoyCommands === 0);
    tryUnlock('rapid-response', metrics.won && metrics.ticks <= 22);

  function finalizeOperation(state, contract) {
    if (!state || !contract || !state.__director || state.__director.finalized) return;
    state.__director.finalized = true;
    const metrics = scoreOperation(state);
    const checks = contractChecks(contract, state, metrics);
    const contractCleared = metrics.won && checks.every((c) => c.pass);
    const medal = medalForScore(metrics.score, contractCleared);
    const date = currentDateKey(), dailyContract = dailyContractFor(date);
    let xpGain = 0, dailyBonus = 0;
    profile.operations += 1;
    if (metrics.won) profile.wins += 1;
    profile.bestScore = Math.max(profile.bestScore, metrics.score);
    const record = ensureContractRecord(contract.id);
    if (contract.custom) record.plays = (record.plays || 0) + 1;
    if (metrics.score > record.bestScore) record.bestScore = metrics.score;
    if (metrics.won && (record.bestTicks === null || metrics.ticks < record.bestTicks)) record.bestTicks = metrics.ticks;
    record.lastAt = new Date().toISOString();
    if (contractCleared) {
      record.clears += 1;
      if (['none','bronze','silver','gold'].indexOf(medal) > ['none','bronze','silver','gold'].indexOf(record.bestMedal)) record.bestMedal = medal;
      const reward = MEDAL_POINTS[medal] + (record.clears === 1 ? 25 : 0);
      xpGain = Math.max(0, reward - record.bestRewardPoints);
      record.bestRewardPoints = Math.max(record.bestRewardPoints, reward);
      if (!contract.custom && contract.id === dailyContract.id && (!profile.daily || profile.daily.date !== date || !profile.daily.claimed)) {
        dailyBonus = 35; xpGain += dailyBonus; profile.daily = { date, contractId:contract.id, claimed:true };
      }
    } else if (metrics.won) {
      xpGain = 12; // A real extraction still gives small progression, but a failed contract cannot farm medals.
    }
    profile.xp += xpGain;

  function patchEngine() {
    const originalEnqueueAction = Game.enqueueAction;
    const originalAppendPath = Game.appendPath;
    const originalLoadSamplePlan = Game.loadSamplePlan;
    const commandCount = (state) => (state && Array.isArray(state.agents) ? state.agents.reduce((sum, agent) => sum + (Array.isArray(agent.plan) ? agent.plan.length : 0), 0) : 0);
    const snapshotPlans = (state) => Game.clonePlans(state);
    const restorePlans = (state, plans) => { if (state && Array.isArray(state.agents)) state.agents.forEach((agent) => { agent.plan = (plans[agent.id] || []).map((command) => ({...command})); }); };
    const budgetReached = (state, contract, extra = 1) => !!(contract.custom && contract.maxCommands !== null && commandCount(state) + extra > contract.maxCommands);
    Game.enqueueAction = function (state, agentId, type, options = {}) {
      const contract = getContract();
      if (type === Game.ACTION.DECOY && contract.noDecoy) return { ok:false, reason:'Шумовая приманка запрещена текущим контрактом.' };
      if (type === Game.ACTION.HACK && contract.noHack) return { ok:false, reason:'Команда HACK запрещена условиями этого испытания.' };
      if (budgetReached(state, contract, 1)) return { ok:false, reason:`Лимит испытания: не более ${contract.maxCommands} команд.` };
      return originalEnqueueAction.call(Game, state, agentId, type, options);
    };
    Game.appendPath = function (state, agentId, target) {
      const contract = getContract();
      if (!contract.custom || contract.maxCommands === null) return originalAppendPath.call(Game, state, agentId, target);
      const before = snapshotPlans(state); const result = originalAppendPath.call(Game, state, agentId, target);
      if (result && result.ok && commandCount(state) > contract.maxCommands) { restorePlans(state, before); return { ok:false, added:0, reason:`Маршрут превышает бюджет ${contract.maxCommands} команд; выбери более короткий путь.` }; }
      return result;
    };
    Game.loadSamplePlan = function (state) {
      const contract = getContract(); const before = snapshotPlans(state); const result = originalLoadSamplePlan.call(Game, state);
      if (budgetReached(state, contract, 0)) { restorePlans(state, before); announce(`Проверочный план слишком длинный для этого испытания (лимит ${contract.maxCommands} команд).`, true); return false; }
      return result === undefined ? true : result;
    };
    const originalClone = Game.cloneStateForSimulation;
    Game.cloneStateForSimulation = function (plans, options = {}) {
      const state = originalClone.call(Game, plans, options);
      if (launchArmed) {
        launchArmed = false; clearTimeout(launchTimer);
        activeContract = getContract(); activeState = state;
        state.__director = { release:RELEASE, contractId:activeContract.id, finalized:false };
        if (activeContract.noDecoy && state.gadgets) state.gadgets.noiseDecoys = 0;
        appendEvent(state, 'system', `${activeContract.custom ? 'BLACKSITE ARCHITECT' : 'BLACKSITE CONTRACT ACTIVE'} · ${activeContract.name}`);
      }
      return state;
    };
    const originalStep = Game.stepSimulation;
    Game.stepSimulation = function (state) {
      const contract = state && state === activeState && state.__director ? getContract(state.__director.contractId) : null;
      if (!contract || state.status !== 'running') return originalStep.call(Game, state);
      if (contract.maxTicks && state.tick >= contract.maxTicks) {
        endAsFailure(state, `${contract.name}: окно извлечения закрыто на T+${contract.maxTicks}.`, 'warning'); finalizeOperation(state, contract); return state;
      }
      if (contract.id === 'signal-lock' && state.tick >= 5 && state.camera.active) {
        endAsFailure(state, 'Signal Lock failed: MIRA did not disable the camera by T+05.', 'warning'); finalizeOperation(state, contract); return state;
      }
      if (contract.noDecoy && state.gadgets) state.gadgets.noiseDecoys = 0;
      const patrolSpeed = contract.id === 'redline' ? 2 : (Number(contract.guardSpeed) || 1);
      if (patrolSpeed > 1 && state.guard && state.guard.mode === 'patrol' && state.guard.investigateTicks <= 0) {
        const route = Game.MISSION.guardRoute;
        if (route && route.length) {
          for (let move = 1; move < patrolSpeed; move += 1) {
            const next = route[state.guard.routeIndex % route.length];
            state.guard.x = next.x; state.guard.y = next.y; state.guard.routeIndex = (state.guard.routeIndex + 1) % route.length;
            const seen = state.agents.find((agent) => Game.manhattan(agent, state.guard) <= 1);
            if (seen) { endAsFailure(state, `${seen.name} was caught by the accelerated patrol.`, 'danger'); finalizeOperation(state, contract); return state; }
          }
        }
      }
      const result = originalStep.call(Game, state);
      if (state.status !== 'running') finalizeOperation(state, contract);
      return result;
    };
  }

  function findArchitectPath(state,start,destination,walls) {
    if(!state||!start||!destination||!Number.isInteger(destination.x)||!Number.isInteger(destination.y))return null;
    const blocked=new Set(walls.map(p=>`${p[0]},${p[1]}`));
    const key=p=>`${p.x},${p.y}`;const dirs=[{x:1,y:0},{x:0,y:1},{x:-1,y:0},{x:0,y:-1}];
    if(blocked.has(key(destination)))return null;
    const startKey=key(start),endKey=key(destination);if(startKey===endKey)return [];
    const dist=new Map([[startKey,0]]),prev=new Map(),queue=[{...start,cost:0}],visited=new Set();
    while(queue.length){queue.sort((a,b)=>a.cost-b.cost);const cur=queue.shift(),curKey=key(cur);if(visited.has(curKey))continue;visited.add(curKey);if(curKey===endKey)break;
      for(const d of dirs){const next={x:cur.x+d.x,y:cur.y+d.y},nk=key(next);if(blocked.has(nk)||!Game.isWalkable(state,next.x,next.y))continue;const extra=Game.isInCameraZone(next.x,next.y)?12:0,nd=cur.cost+1+extra;if(nd<(dist.get(nk)??Infinity)){dist.set(nk,nd);prev.set(nk,curKey);queue.push({...next,cost:nd});}}
    }
    if(!prev.has(endKey))return null;const path=[];let k=endKey;while(k!==startKey){const [x,y]=k.split(',').map(Number);path.push({x,y});k=prev.get(k);if(!k)return null;}return path.reverse();
  }
  function patchArchitectEngine() {
    const baseWalkable=Game.isWalkable,baseFindPath=Game.findPath,baseAppendPath=Game.appendPath,baseEnqueue=Game.enqueueAction,baseClone=Game.cloneStateForSimulation,baseStep=Game.stepSimulation;
    const activeWalls=()=>architectWallsFor();
    Game.isWalkable=function(state,x,y){return baseWalkable.call(Game,state,x,y)&&!architectWallsFor().some(p=>p[0]===x&&p[1]===y);};
    Game.findPath=function(state,start,destination){const walls=architectWallsFor();return walls.length?findArchitectPath(state,start,destination,walls):baseFindPath.call(Game,state,start,destination);};
    Game.appendPath=function(state,agentId,destination){
      const walls=architectWallsFor();if(!walls.length)return baseAppendPath.call(Game,state,agentId,destination);
      const agent=Game.getAgent(state,agentId);if(!agent)return {ok:false,added:0,reason:'Неизвестный оперативник.'};if(state.status!=='planning')return {ok:false,added:0,reason:'Вернись в режим планирования.'};
      const from=Game.getPlannedPosition(state,agentId),path=findArchitectPath(state,from,destination,walls);if(!path)return {ok:false,added:0,reason:'Маршрут заблокирован укреплением или недоступной клеткой.'};
      const contract=getContract();const count=(state.agents||[]).reduce((n,a)=>n+(a.plan||[]).length,0);if(contract.custom&&contract.maxCommands!==null&&count+path.length>contract.maxCommands)return {ok:false,added:0,reason:`Маршрут превышает бюджет ${contract.maxCommands} команд.`};
      for(const p of path)agent.plan.push({type:Game.ACTION.MOVE,x:p.x,y:p.y});return {ok:true,added:path.length,reason:path.length?`Построен обход укрепления: ${path.length} шагов.`:'Оперативник уже находится в этой клетке.'};
    };
    Game.enqueueAction=function(state,agentId,type,options={}){const walls=architectWallsFor();const target=options.target||options;if(type===Game.ACTION.DECOY&&walls.some(p=>p[0]===target.x&&p[1]===target.y))return {ok:false,reason:'Нельзя установить приманку на укрепление.'};return baseEnqueue.call(Game,state,agentId,type,options);};
    Game.cloneStateForSimulation=function(plans){const state=baseClone.call(Game,plans);if(state&&state.__director){const contract=getContract(state.__director.contractId);const walls=architectWallsFor(contract);if(walls.length)state.__architect={terrainWalls:walls.map(p=>[p[0],p[1]]),layoutVersion:1,blockedCommands:0};}return state;};
    Game.stepSimulation=function(state){
      if(!(state&&state.status==='running'&&state.__architect&&Array.isArray(state.__architect.terrainWalls)&&state.__architect.terrainWalls.length)) return baseStep.call(Game,state);
      const walls=state.__architect.terrainWalls.map(p=>[p[0],p[1]]),blocked=new Set(walls.map(p=>`${p[0]},${p[1]}`)),commandIndex=state.tick;
      for(const agent of state.agents||[]){const cmd=agent.plan&&agent.plan[commandIndex];if(cmd&&cmd.type===Game.ACTION.MOVE&&blocked.has(`${cmd.x},${cmd.y}`)){agent.plan[commandIndex]={type:Game.ACTION.WAIT,architectBlocked:true};state.__architect.blockedCommands=(Number(state.__architect.blockedCommands)||0)+1;appendEvent(state,'warning',`${agent.name}: маршрут упёрся в новое укрепление; команда остановлена.`);}}
      const originalGuard=state.guard;
      let pendingX=null, suppressRouteIndex=false;
      state.guard=new Proxy(originalGuard,{
        set(target,prop,value){
          if(prop==='x'){pendingX=Number(value);return true;}
          if(prop==='y'){
            const wanted={x:Number.isFinite(pendingX)?pendingX:target.x,y:Number(value)};pendingX=null;
            const patrol=target.mode==='patrol'&&(Number(target.investigateTicks)||0)<=0;
            if(!patrol){target.x=wanted.x;target.y=wanted.y;return true;}
            const route=Game.MISSION.guardRoute||[];
            if(!route.length){target.x=originalGuard.x;target.y=originalGuard.y;suppressRouteIndex=true;return true;}
            let startIndex=((Number(target.routeIndex)||0)%route.length+route.length)%route.length, nav=null;
            for(let offset=0;offset<route.length;offset++){
              const index=(startIndex+offset)%route.length,goal=route[index];
              if(blocked.has(`${goal.x},${goal.y}`))continue;
              const path=findArchitectPath(state,{x:target.x,y:target.y},{x:goal.x,y:goal.y},walls);
              if(path){nav={index,goal,path,offset};break;}
            }
            if(!nav){
              target.x=originalGuard.x;target.y=originalGuard.y;suppressRouteIndex=true;
              if(!state.__architect.guardHeld){appendEvent(state,'warning','Патруль заблокирован: не найден свободный узел маршрута.');state.__architect.guardHeld=true;}
              return true;
            }
            state.__architect.guardHeld=false;
            const current={x:target.x,y:target.y};
            const next=nav.path.length?nav.path[0]:current;
            target.x=next.x;target.y=next.y;
            // Keep pursuing this reachable node until actually there. Blocked waypoints are skipped.
            const arrived=next.x===nav.goal.x&&next.y===nav.goal.y;
            target.routeIndex=arrived?(nav.index+1)%route.length:nav.index;
            suppressRouteIndex=true;
            const detouring=nav.offset>0 || nav.path.length>1 || (next.x!==wanted.x||next.y!==wanted.y);
            if(detouring){
              const marker=`${nav.index}:${nav.goal.x},${nav.goal.y}`;
              if(state.__architect.lastGuardRouteMarker!==marker){
                state.__architect.guardDetours=(Number(state.__architect.guardDetours)||0)+1;
                appendEvent(state,'system',`Охрана перестроила маршрут вокруг укреплений и идёт к узлу ${nav.index+1}/${route.length}.`);
                state.__architect.lastGuardRouteMarker=marker;
              }
            } else if(arrived){state.__architect.lastGuardRouteMarker=null;}
            return true;
          }
          if(prop==='routeIndex'&&suppressRouteIndex){suppressRouteIndex=false;return true;}
          target[prop]=value;return true;
        }
      });
      try{return baseStep.call(Game,state);} finally {state.guard={...state.guard};}
    };
  }
